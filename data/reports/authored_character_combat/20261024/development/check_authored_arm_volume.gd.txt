## Independent actual indexed-surface collapse guard; not anatomical validation.
## Source shaft is connected geometry, not only high-weight arm vertices: low
## arm-weight inner skin must remain visible to this test.
const PLANES: Array[float]=[.35,.55,.75]
const RAYS:=16
var measurements: Array=[]

func _posed_vertices(surface: Node3D,row: Dictionary,selected: Dictionary={},source_rest:=false) -> PackedVector3Array:
	var matrices: Array[Transform3D]=surface._skin_matrices(row.mesh,row.bind_bones,row.skeleton)
	if source_rest:
		var frame: Transform3D=surface.avatar.global_transform.affine_inverse()*row.skeleton.global_transform
		for binding: int in range(row.bind_bones.size()):matrices[binding]=frame*row.skeleton.get_bone_global_rest(row.bind_bones[binding])*row.mesh.skin.get_bind_pose(binding)
	var result:=PackedVector3Array();result.resize(row.vertices.size())
	var referenced: Variant=row.referenced if selected.is_empty() else selected.keys()
	for index: int in referenced:
		for slot: int in range(4):
			var weight: float=row.weights[index*4+slot]
			if weight>0.:result[index]+=(matrices[row.bones[index*4+slot]]*row.vertices[index])*weight
	return result

func _axis(skeleton: Skeleton3D,helper: Node3D,side: String,source_rest:=false) -> Array[Vector3]:
	var frame: Transform3D=helper.avatar.global_transform.affine_inverse()*skeleton.global_transform
	var upper: Transform3D=skeleton.get_bone_global_rest(helper._find(side+"UpperArm")) if source_rest else skeleton.get_bone_global_pose(helper._find(side+"UpperArm"))
	var lower: Transform3D=skeleton.get_bone_global_rest(helper._find(side+"Forearm")) if source_rest else skeleton.get_bone_global_pose(helper._find(side+"Forearm"))
	return [frame*upper.origin,frame*lower.origin]

func _shaft(surface: Node3D,row: Dictionary,skeleton: Skeleton3D,helper: Node3D,side: String) -> Dictionary:
	var vertices:=_posed_vertices(surface,row,{},true);var ax:=_axis(skeleton,helper,side,true);var direction: Vector3=(ax[1]-ax[0]).normalized();var length: float=ax[0].distance_to(ax[1])
	var outward:=Vector3(signf(ax[0].x),0.,0.);var candidate: Dictionary={};var adjacency: Dictionary={};var source_faces: Array=[];var best_seed:=-1;var best_score:=-INF
	var arm_bind: Dictionary={}
	for binding: int in range(row.bind_bones.size()):
		var title:=String(skeleton.get_bone_name(row.bind_bones[binding]))
		if title in (["Bip01 L UpperArm","Bip01 L Forearm"] if side=="Left" else ["Bip01 R UpperArm","Bip01 R Forearm"]):arm_bind[binding]=true
	for index: int in row.referenced:
		var delta: Vector3=vertices[index]-ax[0];var u: float=delta.dot(direction)/length;var radial: Vector3=delta-direction*delta.dot(direction)
		# The source armpit join is above this strip. Its slit separates the
		# arm tube from the torso; the graph then selects the outer arm seed.
		if u<.27 or u>.94 or radial.length()>.095:continue
		candidate[index]=true;var arm_weight:=0.
		for slot: int in range(4):
			if arm_bind.has(row.bones[index*4+slot]):arm_weight+=row.weights[index*4+slot]
		var score: float=radial.dot(outward)-absf(u-.55)*.03
		if arm_weight>=.8 and score>best_score:best_seed=index;best_score=score
	var indices: PackedInt32Array=row.mesh.mesh.surface_get_arrays(int(row.surface))[Mesh.ARRAY_INDEX]
	for at: int in range(0,indices.size(),3):
		var a:=indices[at];var b:=indices[at+1];var c:=indices[at+2]
		if not(candidate.has(a) and candidate.has(b) and candidate.has(c)):continue
		source_faces.append([a,b,c])
		for edge: Array in [[a,b],[b,c],[c,a]]:
			if not adjacency.has(edge[0]):adjacency[edge[0]]=[]
			if not adjacency.has(edge[1]):adjacency[edge[1]]=[]
			adjacency[edge[0]].append(edge[1]);adjacency[edge[1]].append(edge[0])
	var connected: Dictionary={};var stack: Array=[]
	if best_seed>=0 and adjacency.has(best_seed):stack.append(best_seed);connected[best_seed]=true
	while not stack.is_empty():
		var index: int=stack.pop_back()
		for other: int in adjacency.get(index,[]):
			if not connected.has(other):connected[other]=true;stack.append(other)
	var faces: Array=[]
	for face: Array in source_faces:
		if connected.has(face[0]) and connected.has(face[1]) and connected.has(face[2]):faces.append(face)
	return {"row":row,"faces":faces,"referenced":connected,"source_vertices":vertices,"selection":{"surface":int(row.surface),"side":side,"candidate_vertices":candidate.size(),"connected_vertices":connected.size(),"connected_faces":faces.size(),"seed_index":best_seed,"neutral_axis":ax,"source_strip_fraction":[.27,.94],"cross_section_fractions":PLANES,"max_source_axis_radius_m":.095,"method":"Geometric neutral shaft strip and indexed connected component from outward arm-dominant seed; mixed-weight inner skin retained."}}

func _section(vertices: PackedVector3Array,faces: Array,ax: Array[Vector3],fraction: float) -> Dictionary:
	var direction: Vector3=(ax[1]-ax[0]).normalized();var center: Vector3=ax[0].lerp(ax[1],fraction)
	var first: Vector3=direction.cross(Vector3.FORWARD).normalized()
	if first.length_squared()<.1:first=direction.cross(Vector3.RIGHT).normalized()
	var second: Vector3=direction.cross(first).normalized();var segments: Array=[]
	for face: Array in faces:
		var points: Array[Vector3]=[vertices[face[0]],vertices[face[1]],vertices[face[2]]];var cut: Array[Vector2]=[]
		var distances: Array[float]=[(points[0]-center).dot(direction),(points[1]-center).dot(direction),(points[2]-center).dot(direction)]
		if minf(distances[0],minf(distances[1],distances[2]))>0. or maxf(distances[0],maxf(distances[1],distances[2]))<0.:continue
		for edge: int in range(3):
			var following: int=(edge+1)%3;var a: float=distances[edge];var b: float=distances[following]
			if (a<0.)==(b<0.) or absf(a-b)<1e-10:continue
			var point: Vector3=points[edge].lerp(points[following],a/(a-b))-center
			cut.append(Vector2(point.dot(first),point.dot(second)))
		if cut.size()==2 and cut[0].distance_squared_to(cut[1])>1e-14:segments.append(cut)
	var radii: Array[float]=[];var covered:=0;var minimum:=INF;var maximum:=0.
	for angle: int in range(RAYS):
		var ray:=Vector2(cos(TAU*angle/RAYS),sin(TAU*angle/RAYS));var radius:=INF
		for segment: Array in segments:
			var start: Vector2=segment[0];var edge: Vector2=segment[1]-start;var determinant: float=ray.cross(edge)
			if absf(determinant)<1e-10:continue
			var distance: float=start.cross(edge)/determinant;var segment_fraction: float=start.cross(ray)/determinant
			if distance>.00001 and distance<.16 and segment_fraction>=-.00001 and segment_fraction<=1.00001:radius=minf(radius,distance)
		if is_finite(radius):covered+=1;minimum=minf(minimum,radius);maximum=maxf(maximum,radius);radii.append(radius)
		else:radii.append(-1.)
	var area:=0.
	if covered==RAYS:
		for at: int in range(RAYS):area+=.5*sin(TAU/RAYS)*radii[at]*radii[(at+1)%RAYS]
	return {"fraction":fraction,"axis_center":center,"axis_direction":direction,"rays":RAYS,"covered_rays":covered,"radius_m":radii,"minimum_radius_m":minimum if is_finite(minimum) else 0.,"maximum_radius_m":maximum,"radial_polygon_area_m2":area,"cut_segments":segments.size()}

func _measure(cache: Dictionary,surface: Node3D,skeleton: Skeleton3D,helper: Node3D,side: String,source_rest:=false) -> Array:
	var points:=_posed_vertices(surface,cache.row,cache.referenced,source_rest);var ax:=_axis(skeleton,helper,side,source_rest);var result: Array=[]
	for fraction: float in PLANES:result.append(_section(points,cache.faces,ax,fraction))
	return result
